  <textarea id="inputText" placeholder="Type or paste text here..."></textarea>
  <button onclick="reverseText()">Reverse Text</button>
  <div class="output" id="outputText"></div>

  <script>
    function reverseText() {
      const input = document.getElementById('inputText').value;
      const reversed = input.split('').reverse().join('');
      document.getElementById('outputText').textContent = reversed;
    }


    // Add this function to allow Swift to call it
  function reverseTextFromApp(text) {
    const reversed = text.split('').reverse().join('');
    document.getElementById('outputText').textContent = reversed;
    return reversed; // This value is returned to Swift
    }
  </script>
